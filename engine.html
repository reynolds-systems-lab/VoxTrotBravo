function placeBrushVoxels(pos, isP, overrideSize = brushSize, ctrlQuat = null) { const r = overrideSize - 1; const loopR = (brushShape === 'ribbon') ? Math.ceil(overrideSize * 1.5) : r; // We use a slightly tighter rSq for the outer boundary to keep the shell thin const rSqOuter = (r + 0.4) * (r + 0.4); const cGx = Math.round(pos.x / VOXEL_SIZE); const cGy = Math.round(pos.y / VOXEL_SIZE); const cGz = Math.round(pos.z / VOXEL_SIZE); if (brushShape === 'ribbon' && ctrlQuat) { worldGroup.getWorldQuaternion(tempBrushWQuat); tempBrushInvCtrlQuat.copy(ctrlQuat).invert(); } // Helper to determine if a relative coordinate is within the brush shape const isInside = (dx, dy, dz) => { if (brushShape === 'sphere') { return (dx * dx + dy * dy + dz * dz <= rSqOuter); } else if (brushShape === 'ribbon' && ctrlQuat) { tempBrushLocalOffset.set(dx * VOXEL_SIZE, dy * VOXEL_SIZE, dz * VOXEL_SIZE); tempBrushLocalOffset.applyQuaternion(tempBrushWQuat); tempBrushLocalOffset.applyQuaternion(tempBrushInvCtrlQuat); const halfW = overrideSize * VOXEL_SIZE; const halfH = VOXEL_SIZE * 0.75; const halfD = VOXEL_SIZE * 0.75; return Math.abs(tempBrushLocalOffset.x) <= halfW && Math.abs(tempBrushLocalOffset.y) <= halfH && Math.abs(tempBrushLocalOffset.z) <= halfD; } else { return (Math.abs(dx) <= r && Math.abs(dy) <= r && Math.abs(dz) <= r); } }; // Determine the bounds for color gradients const minX = cGx - loopR, maxX = cGx + loopR; const minY = cGy - loopR, maxY = cGy + loopR; const minZ = cGz - loopR, maxZ = cGz + loopR; for (let dx = -loopR; dx <= loopR; dx++) { for (let dy = -loopR; dy <= loopR; dy++) { for (let dz = -loopR; dz <= loopR; dz++) { // Skip coordinates outside the primary brush volume if (!isInside(dx, dy, dz)) continue; const gx = cGx + dx; const gy = cGy + dy; const gz = cGz + dz; if (isP) { // DRAWING LOGIC if (brushVolume === 'shell' && r > 0) { // A voxel is on the surface if ANY neighbor is outside the shape. // We check the 6 immediate faces. const isSurface = !isInside(dx + 1, dy, dz) || !isInside(dx - 1, dy, dz) || !isInside(dx, dy + 1, dz) || !isInside(dx, dy - 1, dz) || !isInside(dx, dy, dz + 1) || !isInside(dx, dy, dz - 1); if (isSurface) { // Place the shell voxel const color = getDrawColor(gx, gy, gz, minX, maxX, minY, maxY, minZ, maxZ, 0); setVoxel(gx, gy, gz, true, color, matTypes[0]); } else { // AGGRESSIVE CLEARING: // When moving in a continuous stroke, the interior of the current // sphere MUST erase any voxels placed by previous steps to keep // the tube hollow. setVoxel(gx, gy, gz, false, null); } } else { // SOLID MODE: Standard behavior const color = getDrawColor(gx, gy, gz, minX, maxX, minY, maxY, minZ, maxZ, 0); setVoxel(gx, gy, gz, true, color, matTypes[0]); } } else { // ERASING LOGIC: Always remove the voxel setVoxel(gx, gy, gz, false, null); } } } } leftMenuNeedsUpdate = true; }
